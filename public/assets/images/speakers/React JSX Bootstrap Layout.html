✅ Modified JobBoard.jsx (with layout)

jsx
import { useEffect } from "react";
import "./JobBoard.css";
import { Form, Button, Row, Col, Card } from "react-bootstrap";

const JobBoard = () => {
  useEffect(() => {
    const link = document.createElement("link");
    link.rel = "stylesheet";
    link.href =
      "https://static.zohocdn.com/recruit/embed_careers_site/css/v1.1/embed_jobs.css";
    document.head.appendChild(link);

    const script = document.createElement("script");
    script.src =
      "https://static.zohocdn.com/recruit/embed_careers_site/javascript/v1.1/embed_jobs.js";
    script.type = "text/javascript";
    script.onload = () => {
      window.rec_embed_js.load({
        widget_id: "rec_job_listing_div",
        page_name: "Careers",
        source: "CareerSite",
        site: "https://gwcdata.zohorecruit.in",
        brand_color: "#5d5fef",
        empty_job_msg: "No current Openings",
      });
    };
    document.body.appendChild(script);
  }, []);

  return (
    <div className="container my-5">
      <Row>
        {/* Filters */}
        <Col md={4}>
          <h5 className="mb-4">Filters</h5>
          <Card className="p-3">
            <h6>Job Type</h6>
            <Form.Check type="checkbox" label="Full time (1)" />
            <hr />
            <h6>Industry</h6>
            <Form.Check type="checkbox" label="IT Services (1)" />
            <hr />
            <h6>City</h6>
            <Form.Check type="checkbox" label="Hosur (1)" />
            <hr />
            <h6>Country</h6>
            <Form.Check type="checkbox" label="India (1)" />
          </Card>
        </Col>

        {/* Job Listings */}
        <Col md={8}>
          <h3>Join us</h3>
          <h5 className="text-muted mb-4">Current Openings</h5>

          <Form className="d-flex mb-4 gap-3 flex-wrap">
            <Form.Control type="text" placeholder="Job title or skill" />
            <Form.Control type="text" placeholder="City, state/province or country" />
            <Button variant="primary" className="px-4">Search</Button>
          </Form>

          {/* Zoho Jobs Rendered Here */}
          <div
            id="rec_job_listing_div"
            className="job-listing-embed border rounded shadow-sm p-4 bg-white"
          />
        </Col>
      </Row>
    </div>
  );
};

export default JobBoard;
